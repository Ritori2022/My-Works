{{- /* Section-aware prev/next navigation. Uses PrevInSection / NextInSection so navigation stays within the same section. Falls back to global Prev / Next for pages that are not in a named section. */ -}} {{- $inSection := and .CurrentSection .CurrentSection.Title }} {{- if $inSection }} {{- $prev := .PrevInSection }} {{- $next := .NextInSection }} {{- if or $prev $next }}
{{- with $prev }} « 上一篇
{{- .Name -}} {{- end }} {{- with $next }} 下一篇 »
{{- .Name -}} {{- end }}
{{- end }} {{- else }} {{- /* Fallback: global navigation for pages outside named sections */ -}} {{- $pages := where site.RegularPages "Type" "in" site.Params.mainSections }} {{- if and (gt (len $pages) 1) (in $pages . ) }}
{{- with $pages.Next . }} « 上一篇
{{- .Name -}} {{- end }} {{- with $pages.Prev . }} 下一篇 »
{{- .Name -}} {{- end }}
{{- end }} {{- end }}
